function num(id){return parseFloat(document.getElementById(id).value);}

function fixed(x,n=2){return (isNaN(x)||!isFinite(x))?"—":x.toFixed(n);}

let lastReport = "";


function calc(){

  const pH=num('pH'), pCO2=num('pCO2'), HCO3in=num('HCO3in'), SBE=num('SBE'), Lac=num('Lac');

  const pO2=num('pO2'), FiO2=num('FiO2');

  const Na=num('Na'), Cl=num('Cl'), K=num('K'), Ca=num('Ca'), Mg=num('Mg');

  const Alb=num('Alb'), PO4=num('PO4'), Gluc=num('Gluc');

  const naCorrMode = document.getElementById('naCorr').value;

  const chronicMode = document.getElementById('chronic').value;


  const HCO3calc = 0.03 * pCO2 * Math.pow(10, pH - 6.1);

  const HCO3 = HCO3in || HCO3calc;


  let NaCorr = Na;

  if (naCorrMode === 'on' && Gluc > 5.5) NaCorr = Na + 1.6 * ((Gluc - 5.5)/5.5);


  const PF = pO2 / (FiO2/100);

  const AG = NaCorr - (Cl + HCO3);

  const AGcorr = AG + 0.25 * (40 - Alb);

  const deltaAG = AGcorr - 12;

  const deltaHCO3 = 24 - HCO3;

  // ΔΔ считаем только при метаболическом ацидозе (HCO3 < 24)

  const deltaDelta = (HCO3 < 24 && deltaHCO3 > 0) ? deltaAG / deltaHCO3 : NaN;


  const SBE_FW  = 0.3 * (Na - 140);

  const SBE_Cl  = 104 - (Cl * 140 / Na);

  const SBE_Alb = (0.148 * pH - 0.818) * (40 - Alb);

  const SBE_UA  = SBE - SBE_FW - SBE_Cl - SBE_Alb;

  const deltaAP_from_SBE = -SBE_UA;   // ΔАП ≈ −SBE_UA (по учебнику)


  const SIDa = NaCorr + K + Mg - Cl - Lac;

  const AlbGdl = Alb/10;

  const AlbCh = AlbGdl * (0.123*pH - 0.631) * 10;

  const PO4Ch = PO4 * (0.309*pH - 0.469);

  const SIDe = HCO3 + AlbCh + PO4Ch;

  const SIG = SIDa - SIDe;


  // ===== ЧИСЛА =====

  const rows = [

    ["HCO₃⁻ (лабораторный)", fixed(HCO3in,1)+" ммоль/л"],

    ["HCO₃⁻ (расчётный)", fixed(HCO3calc,1)+" ммоль/л"],

    ["Na⁺ (используется)", fixed(NaCorr,1)+" ммоль/л"],

    ["P/F ratio", fixed(PF,0)+" мм рт.ст."],

    ["Anion Gap (АП)", fixed(AG)],

    ["АП коррект. (альбумин)", fixed(AGcorr)],

    ["ΔAG", fixed(deltaAG)],

    ["ΔHCO₃⁻", fixed(deltaHCO3)],

    ["ΔΔ (дельта-разница)", isNaN(deltaDelta) ? "не применимо" : fixed(deltaDelta,2)],

    ["—— SBE-разбивка ——", ""],

    ["SBE (входной)", fixed(SBE,1)],

    ["SBE_FW (эффект воды)", fixed(SBE_FW)],

    ["SBE_Cl (эффект хлора)", fixed(SBE_Cl)],

    ["SBE_Alb (эффект альбумина)", fixed(SBE_Alb)],

    ["SBE_UA (неизмер. анионы)", fixed(SBE_UA)],

    ["ΔАП ≈ −SBE_UA", fixed(deltaAP_from_SBE)],

    ["—— Стюарт ——", ""],

    ["SID_a", fixed(SIDa)],

    ["SID_e", fixed(SIDe)],

    ["SIG", fixed(SIG)],

  ];

  document.getElementById('results').innerHTML = rows.map(r=>`<div class="row"><span>${r[0]}</span><span>${r[1]}</span></div>`).join('');


  // ===== ЛОГИКА =====

  const logs = [];

  const diagnoses = [];

  const explain = [];


  let primary = pH < 7.35 ? "acidemia" : pH > 7.45 ? "alkalemia" : "normal";

  logs.push(`<b>Шаг 1. Вектор pH:</b> ${fixed(pH,3)} → ${primary==='acidemia'?'ацидемия':primary==='alkalemia'?'алкалемия':'норма'}.`);


  const respAcid = pCO2 > 45;

  const respAlk  = pCO2 < 35;

  const metAcid  = HCO3 < 22 || SBE < -2;

  const metAlk   = HCO3 > 26 || SBE > 2;


  // --- Шаг 3: Респираторный компонент ---

  if (respAcid) {

    let mode = chronicMode;

    if (chronicMode === 'auto') mode = (HCO3 > 28) ? 'chronic' : 'acute';

    const expHCO3 = mode === 'acute' ? 24 + 0.1*(pCO2-40) : 24 + 0.35*(pCO2-40);

    logs.push(`<b>Шаг 3. Респираторный ацидоз</b> (${mode==='acute'?'острое':'хроническое'}): ожид. HCO₃⁻ = ${fixed(expHCO3,1)}.`);

    diagnoses.push({name:"Респираторный ацидоз" + (mode==='chronic'?" (хронический)":""), detail:`pCO₂ = ${fixed(pCO2,1)} мм рт.ст. (>45), pH = ${fixed(pH,3)}.`});

    explain.push({

      title:"Респираторный ацидоз",

      why:`pCO₂ = ${fixed(pCO2,1)} мм рт.ст. > 45 → альвеолярная гиповентиляция.`,

      formula:`pCO₂ > 45 мм рт.ст.`,

      details:mode === 'acute'

        ? `Острая форма: HCO₃⁻ растёт на 1 ммоль/л на 10 мм рт.ст. pCO₂ → ожид. HCO₃⁻ = 24 + 0.1×(${fixed(pCO2,1)}−40) = ${fixed(24+0.1*(pCO2-40),1)}.`

        : `Хроническая: почки компенсируют за 3–5 дней → ожид. HCO₃⁻ = 24 + 0.35×(${fixed(pCO2,1)}−40) = ${fixed(24+0.35*(pCO2-40),1)}.`,

      note:`Причины: обструкция ДП, угнетение ЦНС, нейромышечные расстройства.`

    });

  } else if (respAlk) {

    let mode = chronicMode;

    if (chronicMode === 'auto') mode = (HCO3 < 20) ? 'chronic' : 'acute';

    const expHCO3 = mode === 'acute' ? 24 - 0.2*(40-pCO2) : 24 - 0.4*(40-pCO2);

    logs.push(`<b>Шаг 3. Респираторный алкалоз</b> (${mode==='acute'?'острое':'хроническое'}): ожид. HCO₃⁻ = ${fixed(expHCO3,1)}.`);

    diagnoses.push({name:"Респираторный алкалоз" + (mode==='chronic'?" (хронический)":""), detail:`pCO₂ = ${fixed(pCO2,1)} мм рт.ст. (<35), pH = ${fixed(pH,3)}.`});

    explain.push({

      title:"Респираторный алкалоз",

      why:`pCO₂ = ${fixed(pCO2,1)} мм рт.ст. < 35 → гипервентиляция.`,

      formula:`pCO₂ < 35 мм рт.ст.`,

      details:mode === 'acute'

        ? `Острая форма: ожид. HCO₃⁻ = 24 − 0.2×(40−${fixed(pCO2,1)}) = ${fixed(24-0.2*(40-pCO2),1)}.`

        : `Хроническая: ожид. HCO₃⁻ = 24 − 0.4×(40−${fixed(pCO2,1)}) = ${fixed(24-0.4*(40-pCO2),1)}.`,

      note:`Причины: гипоксия, сепсис, боль, тревога, ИВЛ с избыточной вентиляцией.`

    });

  }


  // --- Шаг 4: HAGMA ---

  if (AGcorr > 14) {

    let d = `АП_корр = ${fixed(AGcorr)} мЭкв/л (>14).`;

    if (Lac > 2) d += ` Лактат = ${fixed(Lac,1)} ммоль/л → <b>лактат-ацидоз</b>.`;

    diagnoses.push({name:"Метаболический ацидоз с увеличенным АП" + (Lac>2?" (лактат-ацидоз)":""), detail:d});

    logs.push(`<b>Шаг 4. HAGMA:</b> ${d}`);

    explain.push({

      title:"HAGMA — метаболический ацидоз с увеличенным АП",

      why:`АП_корр = ${fixed(AGcorr)} > 14 → накопление нелетучих кислот.`,

      formula:`АП = Na⁺ − (Cl⁻ + HCO₃⁻) = ${fixed(NaCorr,1)} − (${fixed(Cl,1)} + ${fixed(HCO3,1)}) = ${fixed(AG)}\nАП_корр = АП + 0.25 × (40 − Alb) = ${fixed(AG)} + 0.25 × (40 − ${fixed(Alb,1)}) = ${fixed(AGcorr)}`,

      details:Lac > 2

        ? `Лактат ${fixed(Lac,1)} ммоль/л — ведущий источник HAGMA.`

        : `Источник — эндогенные кислоты (кетоны, лактат, уремия, токсины).`,

      note:`Норма АП: 8–12 мЭкв/л. Норма АП_корр: ≤14.`

    });

  } else {

    logs.push(`<b>Шаг 4. HAGMA:</b> АП_корр = ${fixed(AGcorr)} ≤ 14 — норма.`);

  }


  // --- Шаг 5: SBE_UA / ΔАП ---

  if (deltaAP_from_SBE > 2) {

    diagnoses.push({name:"Метаболический ацидоз по SBE_UA / ΔАП (неизмеряемые анионы)",

      detail:`ΔАП = ${fixed(deltaAP_from_SBE)} ммоль/л (>2) → накопление неизмеряемых анионов.`});

    logs.push(`<b>Шаг 5. SBE_UA → ΔАП:</b> SBE_UA = ${fixed(SBE_UA)}, ΔАП = ${fixed(deltaAP_from_SBE)} (>2) → неизмеряемые анионы.`);

    explain.push({

      title:"Метаболический ацидоз по SBE_UA / ΔАП",

      why:`ΔАП = ${fixed(deltaAP_from_SBE)} > 2 ммоль/л — накопление неизмеряемых анионов.`,

      formula:`SBE_FW  = 0.3 × (Na − 140) = 0.3 × (${fixed(Na,1)} − 140) = ${fixed(SBE_FW)}\nSBE_Cl  = 104 − (Cl × 140 / Na) = 104 − (${fixed(Cl,1)} × 140 / ${fixed(Na,1)}) = ${fixed(SBE_Cl)}\nSBE_Alb = (0.148 × pH − 0.818) × (40 − Alb) = (0.148 × ${fixed(pH,3)} − 0.818) × (40 − ${fixed(Alb,1)}) = ${fixed(SBE_Alb)}\nSBE_UA  = SBE − SBE_FW − SBE_Cl − SBE_Alb = ${fixed(SBE_UA)}\nΔАП ≈ −SBE_UA = ${fixed(deltaAP_from_SBE)}`,

      details:`По учебнику SBE_UA отрицательный, но его модуль отражает ΔАП.`,

      note:`Норма ΔАП: −2…+2. Повышение > 2 — HAGMA по Стюарту.`

    });

  } else {

    logs.push(`<b>Шаг 5. SBE_UA → ΔАП:</b> ΔАП = ${fixed(deltaAP_from_SBE)} ≤ 2 — норма.`);

  }


  // --- Шаг 6: ΔΔ ---

  if (!isNaN(deltaDelta)) {

    if (deltaDelta < 1) {

      diagnoses.push({name:"Метаболический ацидоз с нормальным АП (гиперхлоремический, NAGMA)",

        detail:`ΔΔ = ${fixed(deltaDelta,2)} (<1), HCO₃⁻ = ${fixed(HCO3,1)}, Cl⁻ = ${fixed(Cl,1)} → гиперхлоремия.`});

      logs.push(`<b>Шаг 6. ΔΔ:</b> ${fixed(deltaDelta,2)} (<1) → NAGMA.`);

      explain.push({

        title:"NAGMA — гиперхлоремический метаболический ацидоз",

        why:`ΔΔ = ${fixed(deltaDelta,2)} < 1 при HCO₃⁻ = ${fixed(HCO3,1)} и Cl⁻ = ${fixed(Cl,1)}.`,

        formula:`ΔAG = АП_корр − 12 = ${fixed(AGcorr)} − 12 = ${fixed(deltaAG)}\nΔHCO₃⁻ = 24 − HCO₃⁻ = 24 − ${fixed(HCO3,1)} = ${fixed(deltaHCO3)}\nΔΔ = ΔAG / ΔHCO₃⁻ = ${fixed(deltaAG)} / ${fixed(deltaHCO3)} = ${fixed(deltaDelta,2)}`,

        details:`ΔΔ < 1 → часть ацидоза обусловлена гиперхлоремией.`,

        note:`Причины: диарея, RTA, инфузия NaCl, ацетазоламид. Норма ΔΔ: 1–2.`

      });

    } else if (deltaDelta > 2) {

      diagnoses.push({name:"Сопутствующий метаболический алкалоз (по ΔΔ)",

        detail:`ΔΔ = ${fixed(deltaDelta,2)} (>2) → скрытый метаболический алкалоз.`});

      logs.push(`<b>Шаг 6. ΔΔ:</b> ${fixed(deltaDelta,2)} (>2) → мет. алкалоз.`);

      explain.push({

        title:"Сопутствующий метаболический алкалоз (по ΔΔ)",

        why:`ΔΔ = ${fixed(deltaDelta,2)} > 2 → АП вырос сильнее, чем упал HCO₃⁻.`,

        formula:`ΔΔ = ΔAG / ΔHCO₃⁻ = ${fixed(deltaAG)} / ${fixed(deltaHCO3)} = ${fixed(deltaDelta,2)}`,

        details:`Избыточный прирост АП → есть параллельный алкалоз.`,

        note:`Причины: рвота, диуретики, гипокалиемия, гипохлоремия.`

      });

    } else {

      logs.push(`<b>Шаг 6. ΔΔ:</b> ${fixed(deltaDelta,2)} (1–2) — изолированный HAGMA.`);

    }

  } else {

    logs.push(`<b>Шаг 6. ΔΔ:</b> не применимо (HCO₃⁻ ≥ 24 — нет метаболического ацидоза).`);

  }


  // --- Шаг 7: Метаболический алкалоз (независимо от респираторного!) ---

  if (metAlk) {

    diagnoses.push({name:"Метаболический алкалоз",

      detail:`SBE = ${fixed(SBE,1)} (>2) и/или HCO₃⁻ = ${fixed(HCO3,1)} (>26).`});

    logs.push(`<b>Шаг 7. Метаболический алкалоз:</b> SBE = ${fixed(SBE,1)} > 2.`);

    explain.push({

      title:"Метаболический алкалоз",

      why:`SBE = ${fixed(SBE,1)} > 2 и/или HCO₃⁻ = ${fixed(HCO3,1)} > 26.`,

      formula:`SBE > 2 ммоль/л\nHCO₃⁻ > 26 ммоль/л`,

      details:`Избыток оснований → алкалоз. Компенсация: гиповентиляция (рост pCO₂).`,

      note:`Причины: рвота, диуретики, гипокалиемия, гиперальдостеронизм, избыток HCO₃⁻.`

    });

  }

  let winterInfo = "—";

  let compCheck = [];

  if (metAcid) {
